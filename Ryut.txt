def q(name: str) -> str:
    return "`" + name.replace("`", "``") + "`"


def fqtn(catalog: str, schema: str, table: str) -> str:
    return f"{q(catalog)}.{q(schema)}.{q(table)}"


def cfg_for(table: str) -> dict:
    lookup = {str(k).lower(): v for k, v in TABLE_CONFIG.items()}
    cfg = lookup.get(table.lower(), {})
    if cfg is None:
        return {}
    if not isinstance(cfg, dict):
        raise ValueError(f"Configuration for {table} must be a JSON object.")
    return cfg


def empty_df(schema: str) -> DataFrame:
    return spark.createDataFrame([], schema=schema)


def union_all(frames: Sequence[DataFrame], schema: str) -> DataFrame:
    valid = [x for x in frames if x is not None]
    if not valid:
        return empty_df(schema)
    return reduce(
        lambda a, b: a.unionByName(b, allowMissingColumns=True),
        valid,
    )


def read_table(
    catalog: str,
    schema: str,
    table: str,
    where_clause: str = "",
) -> DataFrame:
    df = spark.table(fqtn(catalog, schema, table))
    return df.where(where_clause) if where_clause else df


def write_single_csv(df: DataFrame, base: str) -> str:
    tmp = base + "_tmp"
    final = base + ".csv"
    dbutils.fs.rm(tmp, True)
    dbutils.fs.rm(final, True)

    (
        df.coalesce(1)
        .write.mode("overwrite")
        .option("header", "true")
        .option("quoteAll", "true")
        .csv(tmp)
    )

    parts = [
        x.path for x in dbutils.fs.ls(tmp)
        if x.name.startswith("part-") and x.name.endswith(".csv")
    ]
    if not parts:
        raise RuntimeError(f"No CSV part file produced for {base}")

    dbutils.fs.mv(parts[0], final)
    dbutils.fs.rm(tmp, True)
    return final


def download_url(path: str) -> Optional[str]:
    prefix = "dbfs:/FileStore/"
    if path.startswith(prefix):
        return "/files/" + path[len(prefix):]
    return None


def classify(field: T.StructField) -> str:
    dt = field.dataType
    if isinstance(
        dt,
        (
            T.ByteType,
            T.ShortType,
            T.IntegerType,
            T.LongType,
            T.FloatType,
            T.DoubleType,
            T.DecimalType,
        ),
    ):
        return "NUMERICAL"
    if isinstance(dt, T.DateType):
        return "DATE"
    if isinstance(dt, T.TimestampNTZType):
        return "TIMESTAMP_NTZ"
    if isinstance(dt, T.TimestampType):
        return "TIMESTAMP"
    return "CATEGORICAL"


def get_common_columns(
    src: DataFrame,
    tgt: DataFrame,
) -> Dict[str, Tuple[T.StructField, T.StructField]]:
    s = {f.name.lower(): f for f in src.schema.fields}
    t = {f.name.lower(): f for f in tgt.schema.fields}
    return {
        name: (s[name], t[name])
        for name in sorted(set(s) & set(t))
    }


def apply_volume(
    df: DataFrame,
    row_limit: int,
    order_columns: Sequence[str],
) -> DataFrame:
    """
    Applies a deterministic row limit when row_limit > 0.
    For reliable comparison, row_order_columns should normally include the key columns.
    """
    if row_limit <= 0:
        return df

    if order_columns:
        return df.orderBy(
            *[F.col(q(c)) for c in order_columns]
        ).limit(row_limit)

    return df.limit(row_limit)


def key_json_expr(alias: str, keys: Sequence[str]) -> F.Column:
    return F.to_json(
        F.struct(
            *[
                F.col(f"{alias}.{q(k)}").alias(k)
                for k in keys
            ]
        ),
        options={"ignoreNullFields": "false"},
    )

def canonical_value(
    col_expr: F.Column,
    data_type: T.DataType,
    timezone_name: str,
) -> F.Column:
    """
    Canonical string representation used in the SHA-256 row hash.
    """
    if isinstance(data_type, T.StringType):
        return col_expr.cast("string")

    if isinstance(data_type, T.BooleanType):
        return col_expr.cast("boolean").cast("string")

    if isinstance(
        data_type,
        (
            T.ByteType,
            T.ShortType,
            T.IntegerType,
            T.LongType,
            T.FloatType,
            T.DoubleType,
            T.DecimalType,
        ),
    ):
        return col_expr.cast("decimal(38,18)").cast("string")

    if isinstance(data_type, T.DateType):
        return F.date_format(col_expr.cast("date"), "yyyy-MM-dd")

    if isinstance(data_type, T.TimestampNTZType):
        utc_ts = F.to_utc_timestamp(
            col_expr.cast("timestamp"),
            timezone_name,
        )
        return F.date_format(
            utc_ts,
            "yyyy-MM-dd'T'HH:mm:ss.SSSSSS'Z'",
        )

    if isinstance(data_type, T.TimestampType):
        return F.date_format(
            col_expr.cast("timestamp"),
            "yyyy-MM-dd'T'HH:mm:ss.SSSSSS'Z'",
        )

    if isinstance(data_type, T.BinaryType):
        return F.base64(col_expr)

    if isinstance(
        data_type,
        (T.ArrayType, T.MapType, T.StructType),
    ):
        return F.to_json(
            col_expr,
            options={"ignoreNullFields": "false"},
        )

    return col_expr.cast("string")


def add_row_hash(
    df: DataFrame,
    columns: Sequence[str],
    timezone_name: str,
    hash_column: str,
) -> DataFrame:
    field_map = {f.name.lower(): f for f in df.schema.fields}

    canonical = []
    for c in columns:
        field = field_map[c.lower()]
        canonical.append(
            F.coalesce(
                canonical_value(
                    F.col(q(field.name)),
                    field.dataType,
                    timezone_name,
                ),
                F.lit("∅"),
            ).alias(field.name)
        )

    payload = F.to_json(
        F.struct(*canonical),
        options={"ignoreNullFields": "false"},
    )
    return (
        df
        .withColumn("__hash_payload", payload)
        .withColumn(hash_column, F.sha2(F.col("__hash_payload"), 256))
        .drop("__hash_payload")
    )

NUMERIC_TYPES = (
    T.ByteType,
    T.ShortType,
    T.IntegerType,
    T.LongType,
    T.FloatType,
    T.DoubleType,
    T.DecimalType,
)


def values_equal(
    src_col: F.Column,
    tgt_col: F.Column,
    src_type: T.DataType,
    classification: str,
) -> F.Column:
    both_null = src_col.isNull() & tgt_col.isNull()
    one_null = src_col.isNull() ^ tgt_col.isNull()

    if isinstance(src_type, NUMERIC_TYPES):
        s = src_col.cast("double")
        t = tgt_col.cast("double")
        abs_diff = F.abs(t - s)
        denominator = F.greatest(
            F.abs(s), F.abs(t), F.lit(1.0)
        )
        allowed = F.greatest(
            F.lit(NUM_ABS_TOL),
            F.lit(NUM_REL_TOL) * denominator,
        )
        equal_non_null = abs_diff <= allowed

    elif classification == "TIMESTAMP_NTZ":
        s = F.to_utc_timestamp(
            src_col.cast("timestamp"),
            SOURCE_TZ,
        ).cast("double")
        t = F.to_utc_timestamp(
            tgt_col.cast("timestamp"),
            TARGET_TZ,
        ).cast("double")
        equal_non_null = F.abs(t - s) <= F.lit(TS_TOL_SEC)

    elif classification == "TIMESTAMP":
        equal_non_null = (
            F.abs(
                tgt_col.cast("timestamp").cast("double")
                - src_col.cast("timestamp").cast("double")
            )
            <= F.lit(TS_TOL_SEC)
        )

    else:
        equal_non_null = src_col.eqNullSafe(tgt_col)

    return (
        F.when(both_null, F.lit(True))
        .when(one_null, F.lit(False))
        .otherwise(equal_non_null)
    )


def display_value(
    col_expr: F.Column,
    data_type: T.DataType,
    timezone_name: str,
) -> F.Column:
    return canonical_value(col_expr, data_type, timezone_name)

def row_hash_compare(
    table: str,
    source_df: DataFrame,
    target_df: DataFrame,
    cfg: dict,
) -> Tuple[DataFrame, DataFrame, DataFrame]:
    keys = [str(x) for x in cfg.get("key_columns", [])]
    if not keys:
        raise ValueError(
            f"{table}: row_hash_enabled requires key_columns."
        )

    src_cols = {c.lower(): c for c in source_df.columns}
    tgt_cols = {c.lower(): c for c in target_df.columns}

    missing_src = [k for k in keys if k.lower() not in src_cols]
    missing_tgt = [k for k in keys if k.lower() not in tgt_cols]
    if missing_src or missing_tgt:
        raise ValueError(
            f"{table}: missing keys; source={missing_src}, target={missing_tgt}"
        )

    # Use exact source spelling as logical key spelling.
    keys = [src_cols[k.lower()] for k in keys]

    excluded = {
        str(x).lower()
        for x in cfg.get("excluded_hash_columns", [])
    }

    common = get_common_columns(source_df, target_df)
    hash_columns = [
        pair[0].name
        for name, pair in common.items()
        if name not in excluded
    ]

    # Keys are used for joining, not as part of change detection.
    non_key_hash_columns = [
        c for c in hash_columns
        if c.lower() not in {k.lower() for k in keys}
    ]

    if not non_key_hash_columns:
        raise ValueError(
            f"{table}: no common non-key columns available for row hashing."
        )

    row_limit = int(
        cfg.get("row_limit", ROW_HASH_DEFAULT_LIMIT)
    )
    order_columns = [
        str(x)
        for x in cfg.get("row_order_columns", keys)
    ]

    source_limited = apply_volume(
        source_df,
        row_limit,
        order_columns,
    )
    target_limited = apply_volume(
        target_df,
        row_limit,
        order_columns,
    )

    source_count = source_limited.count()
    target_count = target_limited.count()

    source_hashed = add_row_hash(
        source_limited,
        non_key_hash_columns,
        SOURCE_TZ,
        "__source_hash",
    ).alias("s")

    target_hashed = add_row_hash(
        target_limited,
        [
            common[c.lower()][1].name
            for c in non_key_hash_columns
        ],
        TARGET_TZ,
        "__target_hash",
    ).alias("t")

    join_condition = reduce(
        lambda a, b: a & b,
        [
            F.col(f"s.{q(src_cols[k.lower()])}").eqNullSafe(
                F.col(f"t.{q(tgt_cols[k.lower()])}")
            )
            for k in keys
        ],
    )

    joined = source_hashed.join(
        target_hashed,
        join_condition,
        "fullouter",
    )

    source_exists = F.col("s.__source_hash").isNotNull()
    target_exists = F.col("t.__target_hash").isNotNull()

    record_status = (
        F.when(source_exists & ~target_exists, "SOURCE_ONLY")
        .when(~source_exists & target_exists, "TARGET_ONLY")
        .when(
            F.col("s.__source_hash") == F.col("t.__target_hash"),
            "MATCH",
        )
        .otherwise("CHANGED")
    )

    # Build a key JSON that works even for source-only / target-only rows.
    key_struct_fields = []
    for key in keys:
        s_key = src_cols[key.lower()]
        t_key = tgt_cols[key.lower()]
        key_struct_fields.append(
            F.coalesce(
                F.col(f"s.{q(s_key)}"),
                F.col(f"t.{q(t_key)}"),
            ).alias(key)
        )

    key_json = F.to_json(
        F.struct(*key_struct_fields),
        options={"ignoreNullFields": "false"},
    )

    row_status = joined.select(
        F.lit(RUN_ID).alias("run_id"),
        F.lit(table).alias("table_name"),
        key_json.alias("key_json"),
        record_status.alias("record_status"),
        F.col("s.__source_hash").alias("source_row_hash"),
        F.col("t.__target_hash").alias("target_row_hash"),
        F.lit(RUN_TS).cast("timestamp").alias("run_ts"),
    )

    counts = (
        row_status.groupBy()
        .agg(
            F.sum(
                F.when(F.col("record_status") == "SOURCE_ONLY", 1).otherwise(0)
            ).alias("source_only_rows"),
            F.sum(
                F.when(F.col("record_status") == "TARGET_ONLY", 1).otherwise(0)
            ).alias("target_only_rows"),
            F.sum(
                F.when(F.col("record_status") == "MATCH", 1).otherwise(0)
            ).alias("equal_hash_rows"),
            F.sum(
                F.when(F.col("record_status") == "CHANGED", 1).otherwise(0)
            ).alias("changed_hash_rows"),
        )
        .first()
        .asDict()
    )

    equal_rows = int(counts["equal_hash_rows"] or 0)
    changed_rows = int(counts["changed_hash_rows"] or 0)
    src_only = int(counts["source_only_rows"] or 0)
    tgt_only = int(counts["target_only_rows"] or 0)
    denominator = max(source_count, target_count, 1)
    match_pct = equal_rows / denominator * 100.0

    summary = spark.createDataFrame(
        [{
            "run_id": RUN_ID,
            "table_name": table,
            "requested_row_limit": row_limit,
            "source_rows_hashed": source_count,
            "target_rows_hashed": target_count,
            "source_only_rows": src_only,
            "target_only_rows": tgt_only,
            "equal_hash_rows": equal_rows,
            "changed_hash_rows": changed_rows,
            "hash_match_pct": match_pct,
            "status": (
                "PASS"
                if src_only == 0 and tgt_only == 0 and changed_rows == 0
                else "FAIL"
            ),
            "run_ts": RUN_TS,
        }],
        schema=ROW_HASH_SUMMARY_SCHEMA,
    )

    # Drill-down only changed rows. Source-only/target-only records are already
    # available in row_status with their keys.
    changed_joined = joined.where(
        source_exists
        & target_exists
        & (F.col("s.__source_hash") != F.col("t.__target_hash"))
    )

    detail_frames: List[DataFrame] = []

    for logical_name, (sf, tf) in common.items():
        if logical_name in excluded:
            continue
        if sf.name.lower() in {k.lower() for k in keys}:
            continue

        src_value = F.col(f"s.{q(sf.name)}")
        tgt_value = F.col(f"t.{q(tf.name)}")
        classification = classify(sf)
        equal_expr = values_equal(
            src_value,
            tgt_value,
            sf.dataType,
            classification,
        )

        detail = (
            changed_joined
            .where(~equal_expr)
            .select(
                F.lit(RUN_ID).alias("run_id"),
                F.lit(table).alias("table_name"),
                key_json.alias("key_json"),
                F.lit(sf.name).alias("mismatch_column"),
                F.when(
                    src_value.isNull() & tgt_value.isNotNull(),
                    F.lit("SOURCE_NULL"),
                )
                .when(
                    src_value.isNotNull() & tgt_value.isNull(),
                    F.lit("TARGET_NULL"),
                )
                .otherwise(F.lit("VALUE_MISMATCH"))
                .alias("mismatch_type"),
                display_value(
                    src_value,
                    sf.dataType,
                    SOURCE_TZ,
                ).alias("source_value"),
                display_value(
                    tgt_value,
                    tf.dataType,
                    TARGET_TZ,
                ).alias("target_value"),
                F.col("s.__source_hash").alias("source_row_hash"),
                F.col("t.__target_hash").alias("target_row_hash"),
                F.lit(RUN_TS).cast("timestamp").alias("run_ts"),
            )
        )
        detail_frames.append(detail)

    mismatch_detail = union_all(
        detail_frames,
        MISMATCH_DETAIL_SCHEMA,
    )

    if MAX_MISMATCH_DETAILS_PER_TABLE > 0:
        mismatch_detail = mismatch_detail.limit(
            MAX_MISMATCH_DETAILS_PER_TABLE
        )

    return summary, row_status, mismatch_detail

def statistical_compare(
    table: str,
    src: DataFrame,
    tgt: DataFrame,
) -> Tuple[DataFrame, DataFrame, DataFrame, DataFrame, DataFrame]:
    src_count = src.count()
    tgt_count = tgt.count()

    count_df = spark.createDataFrame(
        [{
            "run_id": RUN_ID,
            "table_name": table,
            "source_row_count": src_count,
            "target_row_count": tgt_count,
            "difference": tgt_count - src_count,
            "status": "PASS" if src_count == tgt_count else "FAIL",
            "run_ts": RUN_TS,
        }],
        COUNT_SCHEMA,
    )

    common = get_common_columns(src, tgt)

    schema_rows = []
    for name in sorted(
        set(f.name.lower() for f in src.schema.fields)
        | set(f.name.lower() for f in tgt.schema.fields)
    ):
        sf = next(
            (f for f in src.schema.fields if f.name.lower() == name),
            None,
        )
        tf = next(
            (f for f in tgt.schema.fields if f.name.lower() == name),
            None,
        )

        if sf is None:
            status = "TARGET_ONLY_COLUMN"
            cls = classify(tf)
        elif tf is None:
            status = "SOURCE_ONLY_COLUMN"
            cls = classify(sf)
        elif sf.dataType.simpleString() != tf.dataType.simpleString():
            status = "TYPE_MISMATCH"
            cls = classify(sf)
        else:
            status = "PASS"
            cls = classify(sf)

        schema_rows.append({
            "run_id": RUN_ID,
            "table_name": table,
            "column_name": name,
            "source_data_type": sf.dataType.simpleString() if sf else None,
            "target_data_type": tf.dataType.simpleString() if tf else None,
            "classification": cls,
            "status": status,
            "run_ts": RUN_TS,
        })

    schema_df = spark.createDataFrame(schema_rows, SCHEMA_SCHEMA)

    category_frames = []
    numeric_rows = []
    datetime_rows = []

    for _, (sf, tf) in common.items():
        if sf.dataType.simpleString() != tf.dataType.simpleString():
            continue

        cls = classify(sf)

        if cls == "CATEGORICAL":
            s_distinct = src.select(F.col(q(sf.name))).distinct().count()
            t_distinct = tgt.select(F.col(q(tf.name))).distinct().count()

            sc = (
                src.groupBy(
                    F.coalesce(
                        F.col(q(sf.name)).cast("string"),
                        F.lit("<NULL>")
                    ).alias("category_value")
                )
                .count()
                .withColumnRenamed("count", "source_count")
            )
            tc = (
                tgt.groupBy(
                    F.coalesce(
                        F.col(q(tf.name)).cast("string"),
                        F.lit("<NULL>")
                    ).alias("category_value")
                )
                .count()
                .withColumnRenamed("count", "target_count")
            )

            category_frames.append(
                sc.join(tc, "category_value", "full")
                .select(
                    F.lit(RUN_ID).alias("run_id"),
                    F.lit(table).alias("table_name"),
                    F.lit(sf.name).alias("column_name"),
                    F.col("category_value"),
                    F.coalesce(F.col("source_count"), F.lit(0)).cast("long")
                        .alias("source_count"),
                    F.coalesce(F.col("target_count"), F.lit(0)).cast("long")
                        .alias("target_count"),
                    (
                        F.coalesce(F.col("target_count"), F.lit(0))
                        - F.coalesce(F.col("source_count"), F.lit(0))
                    ).cast("long").alias("difference"),
                    F.lit(s_distinct).cast("long")
                        .alias("source_distinct_count"),
                    F.lit(t_distinct).cast("long")
                        .alias("target_distinct_count"),
                    F.when(
                        F.coalesce(F.col("source_count"), F.lit(0))
                        == F.coalesce(F.col("target_count"), F.lit(0)),
                        "PASS",
                    ).otherwise("FAIL").alias("status"),
                    F.lit(RUN_TS).cast("timestamp").alias("run_ts"),
                )
            )

        elif cls == "NUMERICAL":
            def nstats(df, col):
                c = F.col(q(col)).cast("double")
                return df.agg(
                    F.min(c).alias("MIN"),
                    F.max(c).alias("MAX"),
                    F.sum(c).alias("SUM"),
                    F.expr(
                        f"percentile_approx(cast({q(col)} as double), "
                        f"0.5, {MEDIAN_ACCURACY})"
                    ).alias("MEDIAN"),
                ).first().asDict()

            ss = nstats(src, sf.name)
            ts = nstats(tgt, tf.name)

            for stat in ["MIN", "MAX", "SUM", "MEDIAN"]:
                sv, tv = ss[stat], ts[stat]
                if sv is None and tv is None:
                    diff, status = None, "PASS"
                elif sv is None or tv is None:
                    diff, status = None, "FAIL"
                else:
                    diff = abs(float(tv) - float(sv))
                    denom = max(abs(float(sv)), abs(float(tv)), 1.0)
                    allowed = max(NUM_ABS_TOL, NUM_REL_TOL * denom)
                    status = "PASS" if diff <= allowed else "FAIL"

                numeric_rows.append({
                    "run_id": RUN_ID,
                    "table_name": table,
                    "column_name": sf.name,
                    "statistic_name": stat,
                    "source_value": None if sv is None else float(sv),
                    "target_value": None if tv is None else float(tv),
                    "absolute_difference": diff,
                    "status": status,
                    "run_ts": RUN_TS,
                })

        elif cls in {"DATE", "TIMESTAMP_NTZ", "TIMESTAMP"}:
            if cls == "DATE":
                se = F.col(q(sf.name)).cast("date")
                te = F.col(q(tf.name)).cast("date")
            elif cls == "TIMESTAMP_NTZ":
                se = F.to_utc_timestamp(
                    F.col(q(sf.name)).cast("timestamp"), SOURCE_TZ
                )
                te = F.to_utc_timestamp(
                    F.col(q(tf.name)).cast("timestamp"), TARGET_TZ
                )
            else:
                se = F.col(q(sf.name)).cast("timestamp")
                te = F.col(q(tf.name)).cast("timestamp")

            ss = src.agg(F.min(se).alias("MIN"), F.max(se).alias("MAX")).first().asDict()
            ts = tgt.agg(F.min(te).alias("MIN"), F.max(te).alias("MAX")).first().asDict()

            for stat in ["MIN", "MAX"]:
                sv, tv = ss[stat], ts[stat]

                if sv is None and tv is None:
                    seconds, status = None, "PASS"
                elif sv is None or tv is None:
                    seconds, status = None, "FAIL"
                elif cls == "DATE":
                    seconds = (tv.toordinal() - sv.toordinal()) * 86400.0
                    status = "PASS" if sv == tv else "FAIL"
                else:
                    seconds = tv.timestamp() - sv.timestamp()
                    status = "PASS" if abs(seconds) <= TS_TOL_SEC else "FAIL"

                datetime_rows.append({
                    "run_id": RUN_ID,
                    "table_name": table,
                    "column_name": sf.name,
                    "statistic_name": stat,
                    "source_value": None if sv is None else str(sv),
                    "target_value": None if tv is None else str(tv),
                    "difference_seconds": seconds,
                    "status": status,
                    "run_ts": RUN_TS,
                })

    return (
        count_df,
        schema_df,
        union_all(category_frames, CATEGORY_SCHEMA),
        spark.createDataFrame(numeric_rows, NUMERIC_SCHEMA)
            if numeric_rows else empty_df(NUMERIC_SCHEMA),
        spark.createDataFrame(datetime_rows, DATETIME_SCHEMA)
            if datetime_rows else empty_df(DATETIME_SCHEMA),
    )
